## 1. Base Tables and Initial Data

Run the following in Supabase SQL Editor.

```sql
-- Enable UUID extension
create extension if not exists "uuid-ossp";

-- Allow API roles to use the public schema
grant usage on schema public to anon, authenticated;

-- sections table
create table if not exists public.sections (
  key text primary key,
  name_ar text,
  name_en text,
  name_fr text
);

-- subjects table
create table if not exists public.subjects (
  key text primary key,
  name_ar text,
  name_en text,
  name_fr text
);

-- profiles table (students)
create table if not exists public.profiles (
  id uuid primary key default uuid_generate_v4(),
  username text unique,
  name_ar text not null,
  name_en text,
  name_fr text,
  password text not null,
  section_key text references public.sections(key) on delete set null,
  group_number text,
  work_system text default 'monthly',
  subjects jsonb default '[]'::jsonb,
  parent_name text,
  parent_phone text,
  attendance jsonb default '[]'::jsonb,
  photo_url text,
  homework jsonb default '[]'::jsonb
);

-- schedule_slots table
create table if not exists public.schedule_slots (
  day_key text not null,
  time_slot_id text not null,
  assignment jsonb not null default '"reserved"'::jsonb,
  primary key (day_key, time_slot_id)
);

-- Disable Row Level Security on main tables
alter table public.sections disable row level security;
alter table public.subjects disable row level security;
alter table public.profiles disable row level security;
alter table public.schedule_slots disable row level security;

-- Grant full access to anon and authenticated roles
grant all on public.sections to anon, authenticated;
grant all on public.subjects to anon, authenticated;
grant all on public.profiles to anon, authenticated;
grant all on public.schedule_slots to anon, authenticated;
```

## 3. Interactions Counter

```sql
create table if not exists public."EDSschedule_interactions_counts" (
  event_type text primary key,
  count bigint not null default 1,
  created_at timestamptz not null default now(),
  last_updated timestamptz not null default now()
);

create or replace function public.increment_interaction_count(p_event_type text)
returns void
language sql
as $$
  insert into public."EDSschedule_interactions_counts" (event_type, count)
  values (p_event_type, 1)
  on conflict (event_type)
  do update set
    count = "EDSschedule_interactions_counts".count + 1,
    last_updated = now();
$$;

grant execute on function public.increment_interaction_count(text) to anon, authenticated;
```

Extract interactions:

```sql
select event_type, count, created_at, last_updated
from public."EDSschedule_interactions_counts"
order by last_updated desc;
```

## 4. Backup / Photo Backup

```sql
drop table if exists public."EDSschedule_backup";

create table public."EDSschedule_backup" (
  user_id uuid primary key,
  username text not null,
  photos jsonb not null default '[]'::jsonb,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

grant insert, update, select on public."EDSschedule_backup" to anon, authenticated;

create or replace function public.append_photo_backup(
  p_user_id uuid,
  p_username text,
  p_file_name text,
  p_photo_base64 text
)
returns void
language plpgsql
as $$
begin
  insert into public."EDSschedule_backup" (user_id, username, photos)
  values (
    p_user_id,
    p_username,
    jsonb_build_array(
      jsonb_build_object(
        'file_name', p_file_name,
        'photo_base64', p_photo_base64,
        'created_at', now()
      )
    )
  )
  on conflict (user_id)
  do update set
    photos = public."EDSschedule_backup".photos || jsonb_build_array(
      jsonb_build_object(
        'file_name', p_file_name,
        'photo_base64', p_photo_base64,
        'created_at', now()
      )
    ),
    username = excluded.username,
    updated_at = now();
end;
$$;

grant execute on function public.append_photo_backup(uuid, text, text, text) to anon, authenticated;
```

Extract backups:

```sql
select user_id, username, photos, created_at, updated_at
from public."EDSschedule_backup"
order by updated_at desc;
```

## 5. Notifications System

### 5.1 Tables

```sql
create table if not exists public.notification_batches (
  id uuid primary key default uuid_generate_v4(),
  title text not null,
  body text,
  target_type text,
  target_value text,
  created_at timestamptz not null default now(),
  title_ar text,
  title_en text,
  title_fr text,
  body_ar text,
  body_en text,
  body_fr text
);

create table if not exists public.notifications (
  id uuid primary key default uuid_generate_v4(),
  user_id uuid not null references public.profiles(id) on delete cascade,
  batch_id uuid references public.notification_batches(id) on delete cascade,
  title text not null,
  body text,
  created_at timestamptz not null default now(),
  read boolean not null default false,
  title_ar text,
  title_en text,
  title_fr text,
  body_ar text,
  body_en text,
  body_fr text
);

grant all on public.notifications to anon, authenticated;
grant all on public.notification_batches to anon, authenticated;

alter table public.notifications disable row level security;
alter table public.notification_batches disable row level security;
```

### 5.2 Send Notification RPC

```sql
drop function if exists public.send_notification(text, text, text, text);
drop function if exists public.send_notification(text, text, text, text, text, text, text, text, text, text);

create or replace function public.send_notification(
  p_title text,
  p_body text,
  p_target_type text,
  p_target_value text default null,
  p_title_ar text default null,
  p_title_en text default null,
  p_title_fr text default null,
  p_body_ar text default null,
  p_body_en text default null,
  p_body_fr text default null
)
returns uuid
language plpgsql
as $$
declare
  v_batch_id uuid;
  v_user_id uuid;
begin
  insert into public.notification_batches (
    title, body, target_type, target_value,
    title_ar, title_en, title_fr,
    body_ar, body_en, body_fr
  )
  values (
    p_title, p_body, p_target_type, p_target_value,
    p_title_ar, p_title_en, p_title_fr,
    p_body_ar, p_body_en, p_body_fr
  )
  returning id into v_batch_id;

  if p_target_type = 'all' then
    for v_user_id in select id from public.profiles loop
      insert into public.notifications (user_id, batch_id, title, body, title_ar, title_en, title_fr, body_ar, body_en, body_fr)
      values (v_user_id, v_batch_id, p_title, p_body, p_title_ar, p_title_en, p_title_fr, p_body_ar, p_body_en, p_body_fr);
    end loop;

  elsif p_target_type = 'student' then
    insert into public.notifications (user_id, batch_id, title, body, title_ar, title_en, title_fr, body_ar, body_en, body_fr)
    values (p_target_value::uuid, v_batch_id, p_title, p_body, p_title_ar, p_title_en, p_title_fr, p_body_ar, p_body_en, p_body_fr);

  elsif p_target_type = 'group' then
    for v_user_id in
      select id from public.profiles
      where section_key || '-g' || group_number = p_target_value
    loop
      insert into public.notifications (user_id, batch_id, title, body, title_ar, title_en, title_fr, body_ar, body_en, body_fr)
      values (v_user_id, v_batch_id, p_title, p_body, p_title_ar, p_title_en, p_title_fr, p_body_ar, p_body_en, p_body_fr);
    end loop;

  elsif p_target_type = 'section' then
    for v_user_id in
      select id from public.profiles
      where section_key = p_target_value
    loop
      insert into public.notifications (user_id, batch_id, title, body, title_ar, title_en, title_fr, body_ar, body_en, body_fr)
      values (v_user_id, v_batch_id, p_title, p_body, p_title_ar, p_title_en, p_title_fr, p_body_ar, p_body_en, p_body_fr);
    end loop;

  elsif p_target_type = 'day' then
    for v_user_id in
      select p.id
      from public.profiles p
      where exists (
        select 1
        from public.schedule_slots s
        where s.day_key = p_target_value
          and (
            s.assignment ? p.section_key
            or (p.group_number is not null and s.assignment ? (p.section_key || '-g' || p.group_number))
          )
      )
    loop
      insert into public.notifications (user_id, batch_id, title, body, title_ar, title_en, title_fr, body_ar, body_en, body_fr)
      values (v_user_id, v_batch_id, p_title, p_body, p_title_ar, p_title_en, p_title_fr, p_body_ar, p_body_en, p_body_fr);
    end loop;
  end if;

  return v_batch_id;
end;
$$;

grant execute on function public.send_notification(text, text, text, text, text, text, text, text, text, text) to anon, authenticated;
```

## 6. Direct SQL Examples for Sending Notifications

### All students

```sql
insert into public.notifications (user_id, title, body)
select id, 'إشعار عام', 'رسالة لجميع التلاميذ'
from public.profiles;
```

### Single student by UUID

```sql
insert into public.notifications (user_id, title, body)
values ('STUDENT_UUID', 'إشعار خاص', 'رسالة خاصة بتلميذ واحد');
```

### Whole group (e.g., 6base-g1)

```sql
insert into public.notifications (user_id, title, body)
select id, 'إشعار مجموعة', 'رسالة لمجموعة 6base-g1'
from public.profiles
where section_key || '-g' || group_number = '6base-g1';
```

### Whole section (e.g., 6base)

```sql
insert into public.notifications (user_id, title, body)
select id, 'إشعار قسم', 'رسالة لقسم 6base'
from public.profiles
where section_key = '6base';
```

### By day (e.g., Monday)

```sql
insert into public.notifications (user_id, title, body)
select p.id, 'إشعار يوم الاثنين', 'رسالة لمن لديهم حصص يوم الاثنين'
from public.profiles p
where exists (
  select 1
  from public.schedule_slots s
  where s.day_key = 'mon'
    and (
      s.assignment ? p.section_key
      or (p.group_number is not null and s.assignment ? (p.section_key || '-g' || p.group_number))
    )
);
```

### Using RPC

```sql
select public.send_notification(
  'Title here',
  'Body here',
  'all'                 -- target_type: all, student, group, section, day
  --, 'target_value'    -- optional, required for non-all
);
```

limit 20;

-- Enable UUID extension
create extension if not exists "uuid-ossp";

grant usage on schema public to anon, authenticated;

-- sections table
create table if not exists public.sections (
  key text primary key,
  name_ar text,
  name_en text,
  name_fr text
);

-- subjects table
create table if not exists public.subjects (
  key text primary key,
  name_ar text,
  name_en text,
  name_fr text
);

-- profiles table (students)
create table if not exists public.profiles (
  id uuid primary key default uuid_generate_v4(),
  username text unique,
  name_ar text not null,
  name_en text,
  name_fr text,
  password text not null,
  section_key text references public.sections(key) on delete set null,
  group_number text,
  work_system text default 'monthly',
  subjects jsonb default '[]'::jsonb,
  parent_name text,
  parent_phone text,
  attendance jsonb default '[]'::jsonb,
  photo_url text
);

-- schedule_slots table
create table if not exists public.schedule_slots (
  day_key text not null,
  time_slot_id text not null,
  assignment jsonb not null default '"reserved"'::jsonb,
  primary key (day_key, time_slot_id)
);

-- Disable Row Level Security on all tables

Interactions Counting Table and Function
------------------------------------------------------------
Run in Supabase SQL Editor:
------------------------------------------------------------
-- Create the interaction counts table
create table if not exists public."EDSschedule_interactions_counts" (
  event_type text primary key,
  count bigint not null default 1,
  created_at timestamptz not null default now(),
  last_updated timestamptz not null default now()
);

-- Function to increment a counter atomically
create or replace function public.increment_interaction_count(p_event_type text)
returns void
language sql
as $$
  insert into public."EDSschedule_interactions_counts" (event_type, count)
  values (p_event_type, 1)
  on conflict (event_type)
  do update set
    count = "EDSschedule_interactions_counts".count + 1,
    last_updated = now();
$$;

-- Allow anon/authenticated clients to call the function
grant execute on function public.increment_interaction_count(text) to anon, authenticated;

**Purpose:**  
- `EDSschedule_interactions_counts` stores atomic counters for different events (e.g., `app_open`, `student_login`, `pwa_install`, `photo_update`).  
- The RPC function `increment_interaction_count` is called from the frontend using `supabaseClient.rpc('increment_interaction_count', { p_event_type: '...' })` and is safe for anonymous use.

**Extracting interaction data:**
```sql
-- Get all interaction events and their counts
select
  event_type,
  count,
  created_at,
  last_updated
from public."EDSschedule_interactions_counts"
order by last_updated desc;
```

Backup Table and Photo Backup
------------------------------------------------------------
Run in Supabase SQL Editor:
------------------------------------------------------------
-- Drop old backup table (if exists from previous version)
drop table if exists public."EDSschedule_backup";

-- Create new backup table: one row per user, photos stored as JSONB array
create table public."EDSschedule_backup" (
  user_id uuid primary key,
  username text not null,
  photos jsonb not null default '[]'::jsonb,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

-- Allow anon/authenticated clients to insert/update
grant insert, update, select on public."EDSschedule_backup" to anon, authenticated;

-- Function to append a new photo to a user's photos array
create or replace function public.append_photo_backup(
  p_user_id uuid,
  p_username text,
  p_file_name text,
  p_photo_base64 text
)
returns void
language plpgsql
as $$
begin
  insert into public."EDSschedule_backup" (user_id, username, photos)
  values (
    p_user_id,
    p_username,
    jsonb_build_array(
      jsonb_build_object(
        'file_name', p_file_name,
        'photo_base64', p_photo_base64,
        'created_at', now()
      )
    )
  )
  on conflict (user_id)
  do update set
    photos = public."EDSschedule_backup".photos || jsonb_build_array(
      jsonb_build_object(
        'file_name', p_file_name,
        'photo_base64', p_photo_base64,
        'created_at', now()
      )
    ),
    username = excluded.username,
    updated_at = now();
end;
$$;

-- Grant execute to anon/authenticated
grant execute on function public.append_photo_backup(uuid, text, text, text) to anon, authenticated;

**Purpose:**  
- `EDSschedule_backup` stores historical actions; currently used for photo backups.  
- When a user selects a folder in `profile.html`, every image file is immediately sent to Supabase via the `append_photo_backup` RPC.  
- The table stores one row per user, with a `photos` JSONB array containing objects with `file_name`, `photo_base64`, and `created_at`.  
- The function is fire-and-forget in the frontend, never interrupting the UI.

**Extracting backup data:**
```sql
-- Get all backup rows (one per user)
select
  user_id,
  username,
  photos,
  created_at,
  updated_at
from public."EDSschedule_backup"
order by updated_at desc;

-- Get backups as a single JSON array
select json_agg(
  json_build_object(
    'user_id', user_id,
    'username', username,
    'photos', photos,
    'created_at', created_at,
    'updated_at', updated_at
  )
) as backup_data
from public."EDSschedule_backup";

-- Expand photos for a specific user into separate rows
select
  b.user_id,
  b.username,
  jsonb_array_elements(b.photos) as photo
from public."EDSschedule_backup" b
where b.user_id = 'your-user-uuid-here';

## Notifications System
------------------------------------------------------------
Run in Supabase SQL Editor:
------------------------------------------------------------
-- Create notifications table
create table if not exists public.notifications (
  id uuid primary key default uuid_generate_v4(),
  user_id uuid not null references public.profiles(id) on delete cascade,
  title text not null,
  body text,
  created_at timestamptz not null default now(),
  read boolean not null default false
);

-- Grant access
grant select, insert, update, delete on public.notifications to anon, authenticated;

-- Function to send notifications
create or replace function public.send_notification(
  p_title text,
  p_body text,
  p_target_type text,
  p_target_value text default null
)
returns void
language plpgsql
as $$
declare
  v_user_id uuid;
begin
  if p_target_type = 'all' then
    for v_user_id in select id from public.profiles loop
      insert into public.notifications (user_id, title, body)
      values (v_user_id, p_title, p_body);
    end loop;

  elsif p_target_type = 'student' then
    insert into public.notifications (user_id, title, body)
    values (p_target_value::uuid, p_title, p_body);

  elsif p_target_type = 'group' then
    for v_user_id in
      select id
      from public.profiles
      where section_key || '-g' || group_number = p_target_value
    loop
      insert into public.notifications (user_id, title, body)
      values (v_user_id, p_title, p_body);
    end loop;

  elsif p_target_type = 'section' then
    for v_user_id in
      select id
      from public.profiles
      where section_key = p_target_value
    loop
      insert into public.notifications (user_id, title, body)
      values (v_user_id, p_title, p_body);
    end loop;

  elsif p_target_type = 'day' then
    for v_user_id in
      select p.id
      from public.profiles p
      where exists (
        select 1
        from public.schedule_slots s
        where s.day_key = p_target_value
          and (
            s.assignment ? p.section_key
            or (p.group_number is not null and s.assignment ? (p.section_key || '-g' || p.group_number))
          )
      )
    loop
      insert into public.notifications (user_id, title, body)
      values (v_user_id, p_title, p_body);
    end loop;
  end if;
end;
$$;

-- Grant execute
grant execute on function public.send_notification(text, text, text, text) to anon, authenticated;

-- Enable Realtime for notifications table
alter publication supabase_realtime add table public.notifications;

Purpose:

Stores in-app notifications for students.

send_notification RPC sends to all, single, group, section, or by day.

The index.html page uses Supabase Realtime to receive live notification inserts.

Extracting notifications:
select * from public.notifications order by created_at desc limit 100;

insert into public.notifications (user_id, title, body)
select id, 'إشعار قسم', 'رسالة لقسم 6base'

### 5. Send by day (e.g., Monday 'mon')
insert into public.notifications (user_id, title, body)
select p.id, 'إشعار يوم الاثنين', 'رسالة لمن لديهم حصص يوم الاثنين'
from public.profiles p
where exists (
  select 1
  from public.schedule_slots s
  where s.day_key = 'mon'
    and (
      s.assignment ? p.section_key
      or (p.group_number is not null and s.assignment ? (p.section_key || '-g' || p.group_number))
    )
);

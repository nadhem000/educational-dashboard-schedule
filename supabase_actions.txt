create table if not exists public."EDSschedule_interactions_counts" (
  event_type text primary key,
  count bigint not null default 1,
  created_at timestamptz not null default now(),
  last_updated timestamptz not null default now()
);

create or replace function public.increment_interaction_count(p_event_type text)
returns void
language sql
as $$
  insert into public."EDSschedule_interactions_counts" (event_type, count)
  values (p_event_type, 1)
  on conflict (event_type)
  do update set
    count = "EDSschedule_interactions_counts".count + 1,
    last_updated = now();

-- Photo backup
create or replace function public.append_photo_backup(
  p_user_id uuid,
  p_username text,
  p_file_name text,
  p_photo_base64 text
)
returns void
language plpgsql
as $$
begin
  insert into public."EDSschedule_backup" (user_id, username, photos)
  values (
    p_user_id,
    p_username,
    jsonb_build_array(
      jsonb_build_object(
        'file_name', p_file_name,
        'photo_base64', p_photo_base64,
        'created_at', now()
      )
    )
  )
  on conflict (user_id)
  do update set
    photos = public."EDSschedule_backup".photos || jsonb_build_array(
      jsonb_build_object(
        'file_name', p_file_name,
        'photo_base64', p_photo_base64,
        'created_at', now()
      )
    ),
    username = excluded.username,
    updated_at = now();
end;
$$;

grant execute on function public.append_photo_backup(uuid, text, text, text) to anon, authenticated;

-- Forum file backup
create or replace function public.append_forum_file_backup(
  p_user_id uuid,
  p_file_name text,
  p_file_url text
)
returns void
language plpgsql
as $$
begin
  insert into public."EDSschedule_backup" (user_id, username, forum_files)
  values (
    p_user_id,
    '',
    jsonb_build_array(
      jsonb_build_object(
        'file_name', p_file_name,

### 5.2 Send Notification RPC

```sql
drop function if exists public.send_notification(text, text, text, text);
drop function if exists public.send_notification(text, text, text, text, text, text, text, text, text, text);

create or replace function public.send_notification(
  p_title text,
  p_body text,
  p_target_type text,
  p_target_value text default null,
  p_title_ar text default null,
  p_title_en text default null,
  p_title_fr text default null,
  p_body_ar text default null,
  p_body_en text default null,
  p_body_fr text default null
)
returns uuid
language plpgsql
as $$
declare
  v_batch_id uuid;
  v_user_id uuid;
begin
  insert into public.notification_batches (
    title, body, target_type, target_value,
    title_ar, title_en, title_fr,
    body_ar, body_en, body_fr
  )
  values (
    p_title, p_body, p_target_type, p_target_value,
    p_title_ar, p_title_en, p_title_fr,
    p_body_ar, p_body_en, p_body_fr
  )
  returning id into v_batch_id;

  if p_target_type = 'all' then
    for v_user_id in select id from public.profiles loop
      insert into public.notifications (user_id, batch_id, title, body, title_ar, title_en, title_fr, body_ar, body_en, body_fr)
      values (v_user_id, v_batch_id, p_title, p_body, p_title_ar, p_title_en, p_title_fr, p_body_ar, p_body_en, p_body_fr);
    end loop;

  elsif p_target_type = 'student' then
    insert into public.notifications (user_id, batch_id, title, body, title_ar, title_en, title_fr, body_ar, body_en, body_fr)
    values (p_target_value::uuid, v_batch_id, p_title, p_body, p_title_ar, p_title_en, p_title_fr, p_body_ar, p_body_en, p_body_fr);

  elsif p_target_type = 'group' then
    for v_user_id in
      select id from public.profiles
      where section_key || '-g' || group_number = p_target_value
    loop
      insert into public.notifications (user_id, batch_id, title, body, title_ar, title_en, title_fr, body_ar, body_en, body_fr)
      values (v_user_id, v_batch_id, p_title, p_body, p_title_ar, p_title_en, p_title_fr, p_body_ar, p_body_en, p_body_fr);
    end loop;

  elsif p_target_type = 'section' then
    for v_user_id in
      select id from public.profiles
      where section_key = p_target_value
    loop
      insert into public.notifications (user_id, batch_id, title, body, title_ar, title_en, title_fr, body_ar, body_en, body_fr)
      values (v_user_id, v_batch_id, p_title, p_body, p_title_ar, p_title_en, p_title_fr, p_body_ar, p_body_en, p_body_fr);
    end loop;

  elsif p_target_type = 'day' then
    for v_user_id in
      select p.id
      from public.profiles p
      where exists (
        select 1
        from public.schedule_slots s
        where s.day_key = p_target_value
          and (
            s.assignment ? p.section_key
            or (p.group_number is not null and s.assignment ? (p.section_key || '-g' || p.group_number))
          )
      )
    loop
      insert into public.notifications (user_id, batch_id, title, body, title_ar, title_en, title_fr, body_ar, body_en, body_fr)
      values (v_user_id, v_batch_id, p_title, p_body, p_title_ar, p_title_en, p_title_fr, p_body_ar, p_body_en, p_body_fr);
    end loop;
  end if;

  return v_batch_id;
end;
$$;

grant execute on function public.send_notification(text, text, text, text, text, text, text, text, text, text) to anon, authenticated;
```

## 10. Edge Function: `send-push`

Deploy a Supabase Edge Function named `send-push` with the following code. It includes CORS support.

```javascript
import { createClient } from 'npm:@supabase/supabase-js@2';
import webpush from 'npm:web-push@3';

const VAPID_EMAIL = Deno.env.get('VAPID_EMAIL') || 'mailto:dorar.02126@gmail.com';
const VAPID_PRIVATE_KEY = Deno.env.get('VAPID_PRIVATE_KEY') || '';
const VAPID_PUBLIC_KEY = Deno.env.get('VAPID_PUBLIC_KEY') || '';

webpush.setVapidDetails(
  VAPID_EMAIL,
  VAPID_PUBLIC_KEY,
  VAPID_PRIVATE_KEY
);

Deno.serve(async (req) => {
  if (req.method === 'OPTIONS') {
    return new Response(null, {
      status: 204,
      headers: {
        'Access-Control-Allow-Origin': '*',
        'Access-Control-Allow-Methods': 'POST, OPTIONS',
        'Access-Control-Allow-Headers': 'Content-Type',
      },
    });
  }

  if (req.method !== 'POST') {
    return new Response('Method not allowed', {
      status: 405,
      headers: { 'Access-Control-Allow-Origin': '*' }
    });
  }

  let body;
  try {
    body = await req.json();
  } catch {
    return new Response('Invalid JSON', {
      status: 400,
      headers: { 'Access-Control-Allow-Origin': '*' }
    });
  }

  const { title, body: messageBody, target_type, target_value } = body;
  if (!title || !messageBody) {
    return new Response('Missing title or body', {
      status: 400,
      headers: { 'Access-Control-Allow-Origin': '*' }
    });
  }

  const supabase = createClient(
    Deno.env.get('SUPABASE_URL') || '',
    Deno.env.get('SUPABASE_SERVICE_ROLE_KEY') || ''
  );

  let userIds = [];
  if (target_type === 'all' || !target_type) {
    const { data: allUsers, error: usersError } = await supabase
      .from('profiles')
      .select('id');
    if (usersError) return new Response(JSON.stringify({ error: usersError.message }), { status: 500, headers: { 'Access-Control-Allow-Origin': '*' } });
    userIds = allUsers.map(u => u.id);
  } else if (target_type === 'student') {
    userIds = [target_value];
  } else if (target_type === 'group') {
    const { data: users, error: usersError } = await supabase
      .from('profiles')
      .select('id')
      .filter('section_key', 'not.is', null)
      .filter('group_number', 'not.is', null);
    if (usersError) return new Response(JSON.stringify({ error: usersError.message }), { status: 500, headers: { 'Access-Control-Allow-Origin': '*' } });
    userIds = users
      .filter(u => `${u.section_key}-g${u.group_number}` === target_value)
      .map(u => u.id);
  } else if (target_type === 'section') {
    const { data: users, error: usersError } = await supabase
      .from('profiles')
      .select('id')
      .eq('section_key', target_value);
    if (usersError) return new Response(JSON.stringify({ error: usersError.message }), { status: 500, headers: { 'Access-Control-Allow-Origin': '*' } });
    userIds = users.map(u => u.id);
  } else if (target_type === 'day') {
    const { data: profiles, error: profError } = await supabase
      .from('profiles')
      .select('id, section_key, group_number');
    if (profError) return new Response(JSON.stringify({ error: profError.message }), { status: 500, headers: { 'Access-Control-Allow-Origin': '*' } });

    const { data: scheduleSlots, error: schedError } = await supabase
      .from('schedule_slots')
      .select('day_key, assignment')
      .eq('day_key', target_value);
    if (schedError) return new Response(JSON.stringify({ error: schedError.message }), { status: 500, headers: { 'Access-Control-Allow-Origin': '*' } });

    userIds = profiles.filter(p => {
      return scheduleSlots.some(s => {
        const assignment = s.assignment;
        const hasSection = Array.isArray(assignment)
          ? assignment.includes(p.section_key)
          : assignment === p.section_key;
        const hasGroup = p.group_number && (Array.isArray(assignment)
          ? assignment.includes(`${p.section_key}-g${p.group_number}`)
          : assignment === `${p.section_key}-g${p.group_number}`);
        return hasSection || hasGroup;
      });
    }).map(p => p.id);
  }

  if (userIds.length === 0) {
    return new Response(JSON.stringify({ results: [] }), { status: 200, headers: { 'Access-Control-Allow-Origin': '*' } });
  }

  const { data: subscriptions, error } = await supabase
    .from('push_subscriptions')
    .select('*')
    .in('user_id', userIds);

  if (error) {
    return new Response(JSON.stringify({ error: error.message }), { status: 500, headers: { 'Access-Control-Allow-Origin': '*' } });
  }

  const payload = JSON.stringify({ title, body: messageBody, url: './index.html' });
  const results = [];

  for (const sub of subscriptions) {
    try {
      await webpush.sendNotification(sub.subscription, payload);
      results.push({ id: sub.id, status: 'sent' });
    } catch (err) {
      if (err.statusCode === 410 || err.statusCode === 404) {
        await supabase.from('push_subscriptions').delete().eq('id', sub.id);
        results.push({ id: sub.id, status: 'expired' });
      } else {
        results.push({ id: sub.id, status: 'failed', error: err.message });
      }
    }
  }

  return new Response(JSON.stringify({ results }), { status: 200, headers: { 'Access-Control-Allow-Origin': '*' } });
});
```

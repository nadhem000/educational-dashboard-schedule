Database Tables Created

We created four tables in the public schema:

2. subjects
3. profiles
4. schedule_slots

SQL Script Used to Create Tables and Grant Permissions
------------------------------------------------------
Run in Supabase SQL Editor:

------------------------------------------------------------
-- Enable UUID extension
create extension if not exists "uuid-ossp";

-- Allow API roles to use the public schema
grant usage on schema public to anon, authenticated;

-- sections table
create table if not exists public.sections (
  key text primary key,
  name_ar text,
  name_en text,
  name_fr text
);

-- subjects table
create table if not exists public.subjects (
  key text primary key,
  name_ar text,
  name_en text,
  name_fr text
);

-- profiles table (students)
create table if not exists public.profiles (
  id uuid primary key default uuid_generate_v4(),
  username text unique,
  name_ar text not null,
  name_en text,
  name_fr text,
  password text not null,
  section_key text references public.sections(key) on delete set null,
  group_number text,
  work_system text default 'monthly',
  subjects jsonb default '[]'::jsonb,
  parent_name text,
  parent_phone text,
  attendance jsonb default '[]'::jsonb
);

-- schedule_slots table
create table if not exists public.schedule_slots (
  day_key text not null,
  time_slot_id text not null,
  assignment jsonb not null default '"reserved"'::jsonb,
  primary key (day_key, time_slot_id)
);

-- Disable Row Level Security on all four tables
alter table public.sections disable row level security;
alter table public.subjects disable row level security;
alter table public.profiles disable row level security;
alter table public.schedule_slots disable row level security;

-- Grant full access to anon and authenticated roles
grant all on public.sections to anon, authenticated;
grant all on public.subjects to anon, authenticated;
grant all on public.profiles to anon, authenticated;
grant all on public.schedule_slots to anon, authenticated;

## Interactions Counting Table and Function
------------------------------------------------------------
Run in Supabase SQL Editor:
------------------------------------------------------------
-- Create the interaction counts table
create table if not exists public."EDSschedule_interactions_counts" (
  event_type text primary key,
  count bigint not null default 1,
  created_at timestamptz not null default now(),
  last_updated timestamptz not null default now()
);

-- Function to increment a counter atomically
create or replace function public.increment_interaction_count(p_event_type text)
returns void
language sql
as $$
  insert into public."EDSschedule_interactions_counts" (event_type, count)
  values (p_event_type, 1)
  on conflict (event_type)
  do update set
    count = "EDSschedule_interactions_counts".count + 1,
    last_updated = now();
$$;

-- Allow anon/authenticated clients to call the function
grant execute on function public.increment_interaction_count(text) to anon, authenticated;
Purpose:

EDSschedule_interactions_counts stores atomic counters for different events (e.g., app_open, student_login, pwa_install, photo_update).

The RPC function increment_interaction_count is called from the frontend using supabaseClient.rpc('increment_interaction_count', { p_event_type: '...' }) and is safe for anonymous use.
